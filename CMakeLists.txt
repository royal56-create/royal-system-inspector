cmake_minimum_required(VERSION 3.16)
project(royal_system_inspector VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(royal-inspector src/main.c src/system_info.c src/report.c)
target_include_directories(royal-inspector PRIVATE include)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_system tests/test_system.c src/system_info.c)
  target_include_directories(test_system PRIVATE include)
  add_test(NAME system_info_smoke COMMAND test_system)
endif()
install(TARGETS royal-inspector RUNTIME DESTINATION bin)
